### 2.1 创建通用分页模型

**文件位置：** `src/TodoApp.Core/Models/PagedResult.cs`

```csharp
using System.Collections.Generic;

namespace TodoApp.Core.Models
{
    /// <summary>
    /// 通用分页结果模型
    /// 可用于任何需要分页的实体类型
    /// </summary>
    /// <typeparam name="T">实体类型</typeparam>
    public class PagedResult<T>
    {
        #region 属性

        /// <summary>
        /// 当前页的数据列表
        /// </summary>
        public IEnumerable<T> Items { get; set; } = new List<T>();

        /// <summary>
        /// 当前页码（从1开始）
        /// </summary>
        public int CurrentPage { get; set; }

        /// <summary>
        /// 每页记录数
        /// </summary>
        public int PageSize { get; set; }

        /// <summary>
        /// 总记录数
        /// </summary>
        public int TotalCount { get; set; }

        /// <summary>
        /// 总页数（计算属性）
        /// </summary>
        public int TotalPages => (int)Math.Ceiling((double)TotalCount / PageSize);

        /// <summary>
        /// 是否有上一页
        /// </summary>
        public bool HasPreviousPage => CurrentPage > 1;

        /// <summary>
        /// 是否有下一页
        /// </summary>
        public bool HasNextPage => CurrentPage < TotalPages;

        /// <summary>
        /// 第一条记录在全局的序号（用于显示序号）
        /// </summary>
        public int FirstItemIndex => (CurrentPage - 1) * PageSize + 1;

        /// <summary>
        /// 最后一条记录在全局的序号
        /// </summary>
        public int LastItemIndex => Math.Min(CurrentPage * PageSize, TotalCount);

        #endregion

        #region 构造函数

        /// <summary>
        /// 默认构造函数
        /// </summary>
        public PagedResult() { }

        /// <summary>
        /// 带参数的构造函数
        /// </summary>
        public PagedResult(IEnumerable<T> items, int totalCount, int currentPage, int pageSize)
        {
            Items = items;
            TotalCount = totalCount;
            CurrentPage = currentPage;
            PageSize = pageSize;
        }

        #endregion

        #region 辅助方法

        /// <summary>
        /// 创建空的分页结果
        /// </summary>
        public static PagedResult<T> Empty(int currentPage = 1, int pageSize = 10)
        {
            return new PagedResult<T>
            {
                Items = new List<T>(),
                TotalCount = 0,
                CurrentPage = currentPage,
                PageSize = pageSize
            };
        }

        #endregion
    }
}
```

```csharp
namespace TodoApp.Api.Models

        #region 搜索参数

        /// <summary>
        /// 搜索关键词（搜索标题和描述）
        /// </summary>
        [StringLength(200)]
        public string? SearchKeyword { get; set; }

        #endregion

        #region 筛选参数

        /// <summary>
        /// 完成状态筛选：
        /// null=全部, true=已完成, false=未完成
        /// </summary>
        public bool? IsCompleted { get; set; }

        /// <summary>
        /// 优先级筛选：
        /// null=全部, 0=普通, 1=重要, 2=紧急
        /// </summary>
        public int? Priority { get; set; }

        #endregion

        #region 排序参数

        /// <summary>
        /// 排序字段：
        /// CreatedAt, DueDate, Priority, Title
        /// </summary>
        [RegularExpression("CreatedAt|DueDate|Priority|Title",
            ErrorMessage = "无效的排序字段")]
        public string SortBy { get; set; } = "CreatedAt";

        /// <summary>
        /// 排序方向：asc=升序, desc=降序
        /// </summary>
        [RegularExpression("asc|desc", ErrorMessage = "排序方向必须是asc或desc")]
        public string SortOrder { get; set; } = "desc";

        #endregion
    }
```

```csharp
/// <summary>
/// 分页查询实现
/// 使用LINQ构建动态查询
/// </summary>
public async Task<PagedResult<TodoItem>> GetPagedAsync(TodoQueryParameters parameters)
{
    // 构建基础查询
    IQueryable<TodoItem> query = _context.TodoItems;

    // 1️⃣ 应用筛选条件
    query = ApplyFilters(query, parameters);

    // 2️⃣ 应用搜索条件
    query = ApplySearch(query, parameters);

    // 3️⃣ 获取总记录数（在分页前）
    var totalCount = await query.CountAsync();

    // 4️⃣ 应用排序
    query = ApplySorting(query, parameters);

    // 5️⃣ 应用分页
    var items = await query
        .Skip((parameters.PageNumber - 1) * parameters.PageSize)
        .Take(parameters.PageSize)
        .AsNoTracking()  // 只读查询优化
        .ToListAsync();

    // 6️⃣ 构建并返回分页结果
    return new PagedResult<TodoItem>(
        items: items,
        totalCount: totalCount,
        currentPage: parameters.PageNumber,
        pageSize: parameters.PageSize
    );
}

/// <summary>
/// 应用筛选条件
/// </summary>
private static IQueryable<TodoItem> ApplyFilters(IQueryable<TodoItem> query, TodoQueryParameters p)
{
    // 完成状态筛选
    if (p.IsCompleted.HasValue)
    {
        query = query.Where(t => t.IsCompleted == p.IsCompleted.Value);
    }

    // 优先级筛选
    if (p.Priority.HasValue)
    {
        query = query.Where(t => t.Priority == p.Priority.Value);
    }

    return query;
}

/// <summary>
/// 应用搜索条件（模糊匹配）
/// </summary>
private static IQueryable<TodoItem> ApplySearch(IQueryable<TodoItem> query, TodoQueryParameters p)
{
    if (!string.IsNullOrWhiteSpace(p.SearchKeyword))
    {
        var keyword = p.SearchKeyword.Trim();
        query = query.Where(t =>
            t.Title.Contains(keyword) ||      // 标题包含关键词
            (t.Description != null && t.Description.Contains(keyword))  // 或描述包含
        );
    }

    return query;
}

/// <summary>
/// 应用动态排序
/// </summary>
private static IQueryable<TodoItem> ApplySorting(IQueryable<TodoItem> query, TodoQueryParameters p)
{
    var isDescending = p.SortOrder?.ToLower() == "desc";

    return p.SortBy?.ToLower() switch
    {
        "title" when isDescending => query.OrderByDescending(t => t.Title),
        "title" => query.OrderBy(t => t.Title),
        "duedate" when isDescending => query.OrderByDescending(t => t.DueDate),
        "duedate" => query.OrderBy(t => t.DueDate),
        "priority" when isDescending => query.OrderByDescending(t => t.Priority),
        "priority" => query.OrderBy(t => t.Priority),
        _ when isDescending => query.OrderByDescending(t => t.CreatedAt),  // 默认按创建时间
        _ => query.OrderBy(t => t.CreatedAt)
    };
}
```

### 3.2 关键词高亮显示

在表格显示时高亮搜索关键词：

```csharp
// 在Controller或Helper类中添加高亮方法
public static class HtmlHelpers
{
    /// <summary>
    /// 高亮文本中的关键词
    /// </summary>
    public static string HighlightText(string text, string keyword)
    {
        if (string.IsNullOrWhiteSpace(text) || string.IsNullOrWhiteSpace(keyword))
            return text;

        // HTML编码防止XSS
        var encodedText = System.Net.WebUtility.HtmlEncode(text);
        var encodedKeyword = System.Net.WebUtility.HtmlEncode(keyword.Trim());

        // 使用正则替换为带样式的标签
        var highlighted = System.Text.RegularExpressions.Regex.Replace(
            encodedText,
            System.Text.RegularExpressions.Regex.Escape(encodedKeyword),
            match => $"<mark class='bg-warning'>{match.Value}",
            System.Text.RegularExpressions.RegexOptions.IgnoreCase
        );

        return highlighted;
    }
}
```

### 4.1 全局异常处理中间件

创建自定义异常处理中间件：

**文件位置：** `src/TodoApp.Api/Middleware/ExceptionHandlingMiddleware.cs`

```csharp
using System.Net;
using System.Text.Json;
using Microsoft.AspNetCore.Hosting;
using Microsoft.AspNetCore.Http;
using Microsoft.Extensions.Hosting;
using Microsoft.Extensions.Logging;

namespace TodoApp.Api.Middleware
{
    /// <summary>
    /// 全局异常处理中间件
    /// 捕获所有未处理的异常，返回友好的错误响应
    /// </summary>
    public class ExceptionHandlingMiddleware
    {
        private readonly RequestDelegate _next;
        private readonly ILogger<ExceptionHandlingMiddleware> _logger;
        private readonly IHostEnvironment _environment;

        public ExceptionHandlingMiddleware(
            RequestDelegate next,
            ILogger<ExceptionHandlingMiddleware> logger,
            IHostEnvironment environment)
        {
            _next = next;
            _logger = logger;
            _environment = environment;
        }

        public async Task InvokeAsync(HttpContext context)
        {
            try
            {
                // 继续处理请求
                await _next(context);
            }
            catch (Exception ex)
            {
                // 捕获异常并处理
                await HandleExceptionAsync(context, ex);
            }
        }

        private async Task HandleExceptionAsync(HttpContext context, Exception exception)
        {
            // 记录详细错误日志
            _logger.LogError(exception,
                "发生未处理的异常: {Message}\n路径: {Path}\n查询: {QueryString}",
                exception.Message,
                context.Request.Path,
                context.Request.QueryString);

            // 设置响应内容类型
            context.Response.ContentType = "application/json";

            // 根据异常类型设置HTTP状态码
            var response = context.Response;
            var errorResponse = new ErrorResponse();

            switch (exception)
            {
                case KeyNotFoundException:
                    response.StatusCode = (int)HttpStatusCode.NotFound;
                    errorResponse.Message = "请求的资源不存在";
                    break;

                case ArgumentException:
                    response.StatusCode = (int)HttpStatusCode.BadRequest;
                    errorResponse.Message = exception.Message;
                    break;

                case UnauthorizedAccessException:
                    response.StatusCode = (int)HttpStatusCode.Unauthorized;
                    errorResponse.Message = "未授权访问";
                    break;

                default:
                    response.StatusCode = (int)HttpStatusCode.InternalServerError;
                    errorResponse.Message = "服务器内部错误，请稍后重试";

                    // 开发环境显示详细错误信息
                    if (_environment.IsDevelopment())
                    {
                        errorResponse.Detail = exception.StackTrace;
                    }
                    break;
            }

            // 记录异常ID（用于用户反馈）
            errorResponse.ErrorId = Guid.NewGuid().ToString("N")[..8].ToUpper();
            errorResponse.Timestamp = DateTime.Now;

            _logger.LogInformation("错误ID: {ErrorId} - 用户可凭此ID联系支持团队", errorResponse.ErrorId);

            // 序列化并写入响应
            var result = JsonSerializer.Serialize(errorResponse, new JsonSerializerOptions
            {
                PropertyNamingPolicy = JsonNamingPolicy.CamelCase,
                WriteIndented = _environment.IsDevelopment()
            });

            await response.WriteAsync(result);
        }
    }

    /// <summary>
    /// 错误响应模型
    /// </summary>
    public class ErrorResponse
    {
        public string Message { get; set; } = "发生错误";
        public string? Detail { get; set; }
        public string ErrorId { get; set; } = "";
        public DateTime Timestamp { get; set; }
    }
}
```

## 五、性能优化

### 5.1 AsNoTracking优化

对于只读查询，使用`AsNoTracking()`避免变更追踪开销：

```csharp
// 列表查询（只读）
var items = await _context.TodoItems
    .AsNoTracking()  // 不跟踪实体变化
    .ToListAsync();

// 详情查询（只读）
var item = await _context.TodoItems
    .AsNoTracking()
    .FirstOrDefaultAsync(t => t.Id == id);

// 性能对比：
// 有追踪：每次查询额外 ~10-20% 内存和时间
// 无追踪：纯读取操作性能提升明显
```

### 5.2 Select投影减少数据传输

当不需要所有字段时，使用Select投影：

```csharp
// ❌ 低效：获取所有字段
var items = await _context.TodoItems
    .Select(t => new  // 匿名类型投影
    {
        t.Id,
        t.Title,
        t.IsCompleted,
        t.Priority,
        t.DueDate,
        TitlePreview = t.Description != null && t.Description.Length > 50
            ? t.Description.Substring(0, 50) + "..."
            : t.Description
    })
    .ToListAsync();
```

### 5.3 响应压缩中间件

在Program.cs中启用Gzip压缩：

```csharp
// 添加响应压缩服务

```

### 5.4 缓存策略（可选）

对于不常变的数据，可以使用内存缓存：

```csharp
// 注册缓存服务
builder.Services.AddMemoryCache();

// 在Service中使用
public class TodoService : ITodoService
{
    private readonly IMemoryCache _cache;

    public async Task<IEnumerable<TodoItem>> GetAllTodosAsync()
    {
        const string cacheKey = "all_todos";

        // 尝试从缓存获取
        if (!_cache.TryGetValue(cacheKey, out IEnumerable<TodoItem>? todos))
        {
            // 缓存不存在，从数据库获取
            todos = await _repository.GetAllAsync();

            // 设置缓存选项
            var cacheEntryOptions = new MemoryCacheEntryOptions()
                .SetAbsoluteExpiration(TimeSpan.FromMinutes(5))  // 5分钟过期
                .SetSlidingExpiration(TimeSpan.FromMinutes(1))   // 1分钟无访问则过期
                .SetSize(1);                                      // 设置缓存大小

            _cache.Set(cacheKey, todos, cacheEntryOptions);
        }

        return todos!;
    }
}
```

## 九、常见问题及解决方案

### 问题1：分页后搜索条件丢失

**现象：** 翻页后搜索关键词消失

**解决方案：**
确保分页链接保留所有查询参数（已在_PaginationPartial中实现）。

检查URL格式是否正确：
```
✅ /Todo?SearchKeyword=test&IsCompleted=false&PageNumber=2
❌ /Todo?PageNumber=2  （丢失了其他参数）
```

---

### 问题2：大量数据时分页慢

**原因：** COUNT(*)查询在大表上很慢

**优化方案：**

```sql
-- 使用近似计数（如果允许误差）
SELECT COUNT_BIG(*) FROM TodoItems WITH(NOLOCK);

-- 或者使用估算
SELECT rows FROM sys.partitions WHERE object_id = OBJECT_ID('TodoItems');
```

或者考虑使用**游标分页**（基于ID的分页）替代OFFSET分页。

---

### 问题3：移动端表格显示不佳

**解决方案：**

1. 使用响应式表格类：`<table class="table table-responsive">`
2. 小屏幕隐藏次要列（已在CSS中实现）
3. 考虑使用卡片式布局替代表格
