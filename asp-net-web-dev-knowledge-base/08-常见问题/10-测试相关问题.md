## 问题1：单元测试找不到测试方法

### 问题描述
运行 `dotnet test` 时提示找不到任何测试方法。

### 原因及解决

```csharp
// ❌ 错误：缺少 [Fact] 或 [Theory] 特性
public class CalculatorTests
{
    public void TestAdd()  // 缺少 [Fact]！
    {
        var calc = new Calculator();
        Assert.Equal(4, calc.Add(2, 2));
    }
}

// ✅ 正确：使用 xUnit 特性
public class CalculatorTests
{
    [Fact]
    public void Add_TwoPositiveNumbers_ReturnsSum()
    {
        var calc = new Calculator();
        var result = calc.Add(2, 3);
        Assert.Equal(5, result);
    }

    [Theory]
    [InlineData(1, 2, 3)]
    [InlineData(-1, -1, -2)]
    [InlineData(0, 0, 0)]
    public void Add_VariousInputs_ReturnsExpectedResult(int a, int b, int expected)
    {
        var calc = new Calculator();
        Assert.Equal(expected, calc.Add(a, b));
    }
}
```

```bash
# 确保测试项目引用了 xUnit
dotnet add package Microsoft.NET.Test.Sdk
dotnet add package xunit
dotnet add package xunit.runner.visualstudio

# 运行测试
dotnet test --logger "console;verbosity=detailed"
```

## 问题2：Moq Setup 不匹配

### 常见问题与修复

```csharp
// ❌ 错误：Setup返回了错误类型或参数不精确匹配
var mockRepo = new Mock<IUserRepository>();
mockRepo.Setup(r => r.GetByIdAsync(It.IsAny<int>()))
       .ReturnsAsync(new User { Id = 1, Name = "Test" });  // ✅ 正确

// 但调用时使用了不同的方法签名：
var user = await mockRepo.Object.GetByIdAsync(1);  // OK，匹配
var users = await mockRepo.Object.GetAllAsync();     // 返回 null! 因为没Setup

// ❌ 错误：Callback没有正确设置
mockRepo.Setup(r => r.AddAsync(It.IsAny<User>()))
       .Callback<User>(u =>
       {
           if (u == null) throw new ArgumentNullException();
           return Task.CompletedTask;
       });

// ✅ 完整的Moq使用示例
public class UserServiceTests
{
    private readonly Mock<IUserRepository> _mockRepo;
    private readonly Mock<ILogger<UserService>> _mockLogger;
    private readonly UserService _service;

    public UserServiceTests()
    {
        _mockRepo = new Mock<IUserRepository>();
        _mockLogger = new Mock<ILogger<UserService>>();
        _service = new UserService(_mockRepo.Object, _mockLogger.Object);
    }

    [Fact]
    public async Task GetUserById_WhenUserExists_ReturnsUser()
    {
        // Arrange
        var expectedUser = new User { Id = 42, Name = "John" };
        _mockRepo.Setup(r => r.GetByIdAsync(42))
               .ReturnsAsync(expectedUser);

        // Act
        var result = await _service.GetUserByIdAsync(42);

        // Assert
        Assert.NotNull(result);
        Assert.Equal("John", result.Name);
        _mockRepo.Verify(r => r.GetByIdAsync(42), Times.Once);
    }

    [Fact]
    public async Task CreateUser_WhenValid_CallsRepository()
    {
        // Arrange
        User capturedUser = null;
        _mockRepo.Setup(r => r.AddAsync(Capture.In(out capturedUser)))
               .Returns(Task.CompletedTask);

        // Act
        var result = await _service.CreateUserAsync(new CreateUserDto { Name = "Jane" });

        // Assert
        Assert.True(result.IsSuccess);
        Assert.NotNull(capturedUser);
        Assert.Equal("Jane", capturedUser!.Name);
    }
}
```

## 问题3：WebApplicationFactory 启动失败

### 解决方案

```csharp
// 自定义 WebApplicationFactory
public class CustomWebApplicationFactory : WebApplicationFactory<Program>
    protected override IHost CreateHost(IHostBuilder builder)
    {
        builder.ConfigureServices(services =>
        {
            // 替换真实服务为Mock
            services.AddScoped<IEmailService, FakeEmailService>();
            services.AddScoped<IPaymentService, StubPaymentService>();

            // 使用内存数据库进行集成测试
            services.AddDbContext<AppDbContext>(options =>
                options.UseInMemoryDatabase("TestDb"));
        });

        return base.CreateHost(builder);
    }
}

// 使用示例
public class UsersControllerTests : IClassFixture<CustomWebApplicationFactory>
{
    private readonly CustomWebApplicationFactory _factory;

    public UsersControllerTests(CustomWebApplicationFactory factory)
    {
        _factory = factory;
    }

    [Fact]
    public async Task GetUsers_ReturnsSuccess()
    {
        // Arrange
        var client = _factory.CreateClient();

        // Act
        var response = await client.GetAsync("/api/users");

        // Assert
        response.EnsureSuccessStatusCode();
        var users = await response.Content.ReadFromJsonAsync<List<UserDto>>();
        Assert.NotNull(users);
        Assert.NotEmpty(users);
    }

    [Fact]
    public async Task PostUser_WithInvalidData_ReturnsBadRequest()
    {
        var client = _factory.CreateClient();
        
        var content = new StringContent(
            JsonSerializer.Serialize(new { /* 缺少必填字段 */ }),
            Encoding.UTF8,
            "application/json");

        var response = await client.PostAsync("/api/users", content);

        Assert.Equal(HttpStatusCode.BadRequest, response.StatusCode);
    }
}
```

## 问题4：测试之间数据污染（隔离问题）

### 解决方案

```csharp
// 方案1：每个测试使用独立的数据库实例
public class DatabaseIsolationTest : IDisposable
{
    private readonly AppDbContext _dbContext;
    private bool _disposed;

    public DatabaseIsolationTest()
    {
        var options = new DbContextOptionsBuilder<AppDbContext()
            .UseInMemoryDatabase(Guid.NewGuid().ToString())  // 每个测试唯一数据库名
            .Options;

        _dbContext = new AppDbContext(options.Options);
    }

    public void Dispose()
    {
        _dbContext.Database.EnsureDeleted();
    }
}

// 方案2：使用基类 + Collection fixture（xUnit）
public abstract class TestBase : IDisposable
{
    protected AppDbContext DbContext { get; }
    protected IServiceScope Scope { get; }

    protected TestBase()
    {
        var services = new ServiceCollection();
        ConfigureServices(services);  // 配置测试服务
        
        var provider = services.BuildServiceProvider(validateScopes: true);
        Scope = provider.CreateScope();
        DbContext = Scope.ServiceProvider.GetRequiredService<AppDbContext>();
        
        // 确保数据库已创建
        DbContext.Database.EnsureCreated();
    }

    protected virtual void ConfigureServices(IServiceCollection services)
    {
        services.AddDbContext<AppDbContext>(o =>
            o.UseInMemoryDatabase($"TestDb_{Guid.NewGuid()}"));
    }

    public virtual void Dispose()
    {
        DbContext?.Database.EnsureDeleted();
        Scope?.Dispose();
    }
}

// 具体测试类继承基类
public class OrderServiceTests : TestBase
{
    private OrderService _service;

    public OrderServiceTests() : base()
    {
        _service = new OrderService(DbContext);
    }

    [Fact]
    public async Task CreateOrder_WhenStockAvailable_Succeeds()
    {
        // Arrange: 在隔离的数据库中准备数据
        var product = new Product { Id = 1, StockQuantity = 10 };
        DbContext.Products.Add(product);
        await DbContext.SaveChangesAsync();

        // Act
        var result = await _service.CreateOrderAsync(new CreateOrderDto 
        { 
            ProductId = 1, 
            Quantity = 2,
            UserId = 100 
        });

        // Assert: 验证结果
        Assert.True(result.IsSuccess);
        var order = DbContext.Orders.FirstOrDefault(o => o.UserId == 100);
        Assert.NotNull(order);
    }
}
```

## 问题5：异步测试死锁

### 解决方案

```csharp
// ❌ 错误：在同步上下文中调用异步方法
[Fact]
public void BadAsyncTest()
{
    var result = _service.GetDataAsync().Result;  // 可能死锁！
    Assert.NotNull(result);
}

// ✅ 正确：使测试方法本身异步
[Fact]
public async Task GoodAsyncTest()
{
    var result = await _service.GetDataAsync();
    Assert.NotNull(result);
}

// ✅ 对于必须同步的遗留代码测试
[Fact]
public void LegacySyncTest()
{
    // 使用线程池包装异步调用
    var result = Task.Run(() => _service.GetDataAsync()).GetAwaiter().GetResult();
    Assert.NotNull(result);
}
```

## 总结

| 测试类型 | 工具/框架 | 关键要点 |
|---------|----------|---------|
| 单元测试 | xUnit/NUnit + Moq | 隔离依赖、验证交互 |
| 集成测试 | WebApplicationFactory | 内存数据库、真实HTTP管道 |
| E2E测试 | Playwright/Selenium | 真实浏览器、完整流程 |
| 数据隔离 | InMemory DB / Respawn | 每个测试独立环境 |
| 异步测试 | async/await | 不要阻塞异步调用 |

**测试原则**：
1. **AAA模式**：Arrange → Act → Assert
2. **测试命名**：MethodName_Scenario_ExpectedBehavior
3. **一个测试只测一件事**
4. **使用有意义的断言消息**：`Assert.Equal(expected, actual, "应该相等")`
5. **测试覆盖率 > 80%** 作为质量门槛
