# 控制流语句

> **学习目标**：掌握C#中的条件判断、循环语句和跳转语句，能够编写具有逻辑判断和重复执行能力的程序
>
> **前置知识**：变量与数据类型（建议先学习上一节）
>
> **预计用时**：40-60分钟

---

## 一、为什么需要控制流？

想象一下你在开发一个电商网站：

- 用户登录时，需要**判断**账号密码是否正确
- 显示商品列表时，需要**循环**遍历所有商品
- 处理订单时，根据支付状态**选择**不同的处理逻辑
- 搜索商品时，找到匹配项就**跳出**循环

没有控制流，程序只能从上到下按顺序执行，无法做出任何决策或重复操作。控制流语句让程序变得"智能"起来！

## 二、条件判断语句

### 2.1 if-else 语句

**基本语法：**

```csharp
if (条件表达式)
{
    // 条件为true时执行的代码
}
else if (其他条件)
{
    // 其他条件为true时执行
}
else
{
    // 所有条件都不满足时执行
}
```

### 2.2 switch 语句

```csharp

{
    case 值1:
        // 执行代码
        break;
    case 值2:
        // 执行代码
        break;
    default:
        // 默认情况
        break;
}
```

```csharp
public class DiscountCalculator
{
    // 传统switch语句
    public decimal GetDiscountTraditional(string customerType)
    {
        switch (customerType.ToLower())
        {
            case "vip":
                return 0.20m;
            case "member":
                return 0.10m;
            case "new":
                return 0.05m;
            default:
                return 0m;
        }
    }

    // 使用switch表达式（更简洁）
    public decimal GetDiscountExpression(string customerType) => customerType?.ToLower() switch
    {
        "vip" => 0.20m,
        "member" => 0.10m,
        "new" => 0.05m,
        _ => 0m          // _ 表示默认值
    };

    // 带条件的switch表达式
    public string GetMessage(string userType, decimal orderAmount) => (userType, orderAmount) switch
    {
        ("vip", > 10000) => "尊贵的VIP客户，您已升级为钻石会员！",
        ("vip", _) => "感谢VIP客户的信任",
        ("member", > 5000) => "恭喜获得额外9折优惠",
        ("new", _) => "欢迎新用户，首单立减20元",
        (_, > 1000) => "满1000免运费",
        _ => "感谢您的购买"
    };

    public void TestDiscounts()
    {
        decimal price = 999.00m;

        Console.WriteLine("=== 折扣计算器 ===\n");
        Console.WriteLine($"原价: ¥{price:F2}\n");

        string[] types = { "VIP", "Member", "New", "Guest" };
        foreach (var type in types)
        {
            decimal discount = GetDiscountExpression(type);
            decimal finalPrice = price * (1 - discount);

            Console.WriteLine($"{type,-8} | 折扣: {discount * 100,3}% | 实付: ¥{finalPrice,8:F2}");
        }

        Console.WriteLine("\n=== 客户消息测试 ===\n");
        Console.WriteLine(GetMessage("vip", 15000));
        Console.WriteLine(GetMessage("member", 6000));
        Console.WriteLine(GetMessage("new", 200));
        Console.WriteLine(GetMessage("guest", 1200));
    }
}
```

```csharp
public class MultiplicationTableGenerator
{
    public void GenerateTable(int size)
    {
        Console.WriteLine($"=== {size}x{size} 乘法表 ===\n");

        // 打印表头
        Console.Write("     ");  // 对齐用
        for (int i = 1; i <= size; i++)
        {
            Console.Write($"{i,5}", i);  // 宽度为5，右对齐
        }
        Console.WriteLine();
        Console.WriteLine(new string('-', 5 + size * 5));

        // 打印每一行
        for (int row = 1; row <= size; row++)
        {
            Console.Write($"{row,2} |");  // 行号

            for (int col = 1; col <= size; col++)
            {
                int product = row * col;
                Console.Write($"{product,5}");
            }
            Console.WriteLine();  // 换行
        }
    }

    public void GenerateSpecificTable(int number)
    {
        Console.WriteLine($"\n=== {number} 的乘法口诀 ===\n");

        for (int i = 1; i <= 9; i++)
        {
            Console.WriteLine($"{number} x {i} = {number * i,2}");
        }
    }
}
```

#### 示例6：批量数据处理

```csharp
public class DataBatchProcessor
{
    public void ProcessOrders(List<int> orderIds)
    {
        Console.WriteLine("=== 批量订单处理 ===\n");
        int total = orderIds.Count;
        int successCount = 0;
        int failCount = 0;

        for (int i = 0; i < total; i++)
        {
            int orderId = orderIds[i];
            bool success = ProcessSingleOrder(orderId, i + 1, total);

            if (success)
                successCount++;
            else
                failCount++;

            // 每10个显示一次进度
            if ((i + 1) % 10 == 0 || i == total - 1)
            {
                double progress = (double)(i + 1) / total * 100;
                Console.WriteLine($"\n进度: [{new string('█', (int)(progress / 5))}" +
                                $"{new string('░', 20 - (int)(progress / 5))}] {progress:F1}%");
            }

            // 模拟处理延迟
            Thread.Sleep(50);  // 实际项目中不要这样用！
        }

        Console.WriteLine($"\n处理完成！成功: {successCount}, 失败: {failCount}, 成功率: {(double)successCount / total:P1}");
    }

    private bool ProcessSingleOrder(int orderId, int current, int total)
    {
        // 模拟95%成功率
        Random random = new Random();
        bool success = random.NextDouble() > 0.05;

        string status = success ? "✓" : "✗";
        Console.Write($"[{current,3}/{total}] 订单 #{orderId:D4} ... {status}");

        if (!success)
            Console.Write(" (库存不足)");

        Console.WriteLine();

        return success;
    }

    public void GenerateReportData()
    {
        Console.WriteLine("\n=== 月度销售报告数据生成 ===\n");

        // 生成12个月的销售数据
        decimal[] monthlySales = new decimal[12];
        string[] months = { "一月", "二月", "三月", "四月", "五月", "六月",
                           "七月", "八月", "九月", "十月", "十一月", "十二月" };

        Random random = new Random();

        for (int i = 0; i < 12; i++)
        {
            // 模拟销售额：基础值+随机波动+季节因素
            decimal baseSales = 50000m;
            decimal seasonalFactor = 1 + (decimal)Math.Sin((i - 2) * Math.PI / 6) * 0.3m;
            decimal randomFactor = (decimal)(random.NextDouble() * 0.4 + 0.8);  // 80%-120%
            monthlySales[i] = baseSales * seasonalFactor * randomFactor;
        }

        // 输出报表
        Console.WriteLine($"{"月份",-8} {"销售额",15} {"环比增长",12} {"占比",10}");
        Console.WriteLine(new string('-', 48));

        decimal yearTotal = monthlySales.Sum();

        for (int i = 0; i < 12; i++)
        {
            decimal growth = i == 0 ? 0 :
                           (monthlySales[i] - monthlySales[i - 1]) / monthlySales[i - 1] * 100;
            decimal percentage = monthlySales[i] / yearTotal * 100;

            string growthStr = i == 0 ? "-" : $"{growth:+F1;-F1;0}%";

            Console.WriteLine($"{months[i],-8} ¥{monthlySales[i],12:N2} {growthStr,11} {percentage,9:F1}%");
        }

        Console.WriteLine(new string('-', 48));
        Console.WriteLine($"{"年度总计",-8} ¥{yearTotal,12:N2} {"",12} {"100.0%",9}");
    }
}

// 测试

// 生成50个订单ID进行批量处理

batchProcessor.ProcessOrders(orders);

Console.WriteLine();
batchProcessor.GenerateReportData();
```

#### 示例7：用户登录重试机制

```csharp
public class LoginSystem
{
    private const int MaxAttempts = 3;
    private const int LockoutMinutes = 30;

    public void SimulateLogin()
    {
        Console.WriteLine("=== 登录模拟系统 ===\n");

        int attempts = 0;
        bool isLoggedIn = false;

        // 正确的用户名和密码（实际项目应该从数据库读取）
        string correctUsername = "admin";
        string correctPassword = "123456";

        while (attempts < MaxAttempts && !isLoggedIn)
        {
            attempts++;

            Console.WriteLine($"--- 第 {attempts}/{MaxAttempts} 次尝试 ---");

            // 模拟用户输入（这里直接赋值用于演示）
            string inputUsername = attempts == 2 ? "admin" : "user";  // 第2次才输入正确
            string inputPassword = attempts == 2 ? "123456" : "wrong"; // 第2次才输入正确

            Console.WriteLine($"输入用户名: {inputUsername}");
            Console.WriteLine($"输入密码: {'*' * inputPassword.Length}");

            // 验证逻辑
            if (string.IsNullOrEmpty(inputUsername) || string.IsNullOrEmpty(inputPassword))
            {
                Console.WriteLine("❌ 错误：用户名和密码不能为空\n");
                continue;  // 跳过本次循环，不增加失败计数
            }

            if (inputUsername == correctUsername && inputPassword == correctPassword)
            {
                isLoggedIn = true;
                Console.WriteLine("\n✅ 登录成功！");
                Console.WriteLine($"欢迎使用系统，{correctUsername}！");
                ShowDashboard(correctUsername);
            }
            else
            {
                int remaining = MaxAttempts - attempts;
                Console.WriteLine($"\n❌ 用户名或密码错误");
                Console.WriteLine($"   剩余尝试次数: {remaining}");

                if (remaining > 0)
                {
                    Console.WriteLine("提示: 请检查大小写和空格\n");
                }
            }
        }

        if (!isLoggedIn)
        {
            Console.WriteLine($"\n========================================");
            Console.WriteLine("❌ 登录失败次数过多，账户已被锁定");
            Console.WriteLine($"   请在 {LockoutMinutes} 分钟后重试");
            Console.WriteLine("   或联系管理员解锁");
            Console.WriteLine("========================================");
        }
    }

        Console.WriteLine($"在线用户数: {new Random().Next(50, 200)}");
        Console.WriteLine("待处理任务: 3条");
        Console.WriteLine("系统状态: 🟢 正常运行");
    }
}
```

```csharp
public class GuessNumberGame
{
    public void PlayGame(int maxNumber = 100, int maxGuesses = 7)
    {
        Console.WriteLine("=== 猜数字游戏 ===");
        Console.WriteLine($"我已经想好了一个 1 到 {maxNumber} 之间的整数");
        Console.WriteLine($"你有 {maxGuesses} 次机会来猜它\n");

        Random random = new Random();
        int targetNumber = random.Next(1, maxNumber + 1);
        int guesses = 0;
        bool isGuessed = false;

        List<int> guessHistory = new List<int>();

        while (guesses < maxGuesses && !isGuessed)
        {
            guesses++;
            Console.WriteLine($"--- 第 {guesses}/{maxGuesses} 次 ---");

            // 模拟玩家猜测（随机生成用于演示）
            int guess;
            if (guesses < maxGuesses)
            {
                guess = random.Next(1, maxNumber + 1);
                // 避免重复猜相同的数字
                while (guessHistory.Contains(guess))
                    guess = random.Next(1, maxNumber + 1);
            }
            else
            {
                guess = targetNumber;  // 最后一次一定猜对（演示用）
            }

            guessHistory.Add(guess);
            Console.WriteLine($"你猜的数字是: {guess}");

            if (guess == targetNumber)
            {
                isGuessed = true;
                Console.WriteLine($"\n🎉 恭喜你！答案就是 {targetNumber}！");
                Console.WriteLine($"你用了 {guesses} 次猜中了！");
            }
            else if (guess < targetNumber)
            {
                Console.WriteLine("💡 提示: 太小了，再大一点！");
            }
            else
            {
                Console.WriteLine("💡 提示: 太大了，再小一点！");
            }

            // 显示历史记录
            Console.WriteLine($"已猜过的数字: [{string.Join(", ", guessHistory)}]\n");
        }

        if (!isGuessed)
        {
            Console.WriteLine($"😢 很遗憾，你没有在 {maxGuesses} 次内猜中");
            Console.WriteLine($"正确答案是: {targetNumber}");
        }

        // 游戏统计
        ShowGameStats(guesses, isGuessed, maxGuesses);
    }

    private void ShowGameStats(int guesses, bool won, int maxGuesses)
    {
        Console.WriteLine("\n=== 游戏统计 ===");
        Console.WriteLine($"结果: {(won ? "胜利 ✓" : "失败 ✗")}");
        Console.WriteLine($"使用次数: {guesses}/{maxGuesses}");

        if (won)
        {
            string rating = guesses switch
            {
                1 => "🏆 运气爆棚！一击命中！",
                <= 3 => "⭐ 非常优秀！",
                <= 5 => "👍 表现不错！",
                _ => "💪 继续努力！"
            };
            Console.WriteLine($"评价: {rating}");
        }
    }
}
```

```csharp
        // 2. 必须包含大写字母

        // 4. 必须包含数字
```

```csharp
public class CalendarGenerator

    public void GenerateMonthlyCalendar(int year, int month)
    {
        Console.WriteLine($"\n{'=',36}");
        Console.WriteLine($"       {year}年 {month}月");
        Console.WriteLine($"{'=',36}");
        Console.WriteLine(" 日 一 二 三 四 五 六 ");
        Console.WriteLine($"{'-',34}");

        DateTime firstDay = new DateTime(year, month, 1);
        int daysInMonth = DateTime.DaysInMonth(year, month);
        int startDayOfWeek = (int)firstDay.DayOfWeek;  // 0=周日, 1=周一...

        // 调整为周一起始（可选）
        // startDayOfWeek = startDayOfWeek == 0 ? 6 : startDayOfWeek - 1;

        int dayCounter = 1;
        bool printedAllDays = false;

        // 外层循环：行（最多6行）
        for (int week = 0; week < 6 && !printedAllDays; week++)
        {
            // 内层循环：列（一周7天）
            for (int dayOfWeek = 0; dayOfWeek < 7; dayOfWeek++)
            {
                // 在第一天之前打印空白
                if (week == 0 && dayOfWeek < startDayOfWeek)
                {
                    Console.Write("    ");
                }
                else if (dayCounter <= daysInMonth)
                {
                    // 打印日期
                    DateTime currentDate = new DateTime(year, month, dayCounter);

                    // 今天高亮显示
                    if (currentDate.Date == DateTime.Today.Date)
                    {
                        Console.ForegroundColor = ConsoleColor.Green;
                        Console.Write($"[{dayCounter,2}]");
                        Console.ResetColor();
                    }
                    else if (dayOfWeek == 0 || dayOfWeek == 6)  // 周末
                    {
                        Console.ForegroundColor = ConsoleColor.Red;
                        Console.Write($"{dayCounter,4}");
                        Console.ResetColor();
                    }
                    else
                    {
                        Console.Write($"{dayCounter,4}");
                    }

                    dayCounter++;

                    if (dayCounter > daysInMonth)
                    {
                        printedAllDays = true;
                    }
                }
                else
                {
                    Console.Write("    ");
                }
            }
            Console.WriteLine();  // 换行到下一周
        }

        Console.WriteLine($"{'-',34}");
        Console.WriteLine($"今天: {DateTime.Now:yyyy年MM月dd日 dddd}");
    }

    {
```

## 六、最佳实践总结

### 6.1 选择合适的控制流结构

| 场景 | 推荐语句 | 原因 |
|------|----------|------|
| 二选一判断 | `if-else` | 最直观 |
| 多个离散值匹配 | `switch` | 更清晰 |
| 固定次数循环 | `for` | 控制力强 |
| 遍历集合 | `foreach` | 最简洁 |
| 不确定次数 | `while` | 灵活 |
| 至少执行一次 | `do-while` | 保证执行 |

### 6.2 代码质量检查清单

- [ ] 避免超过3层的嵌套（考虑提取方法）
- [ ] `switch`是否有`default`分支
- [ ] 循环是否有明确的终止条件（避免死循环）
- [ ] 是否使用了有意义的变量名作为条件
- [ ] `break`和`continue`的使用是否必要且清晰
- [ ] 复杂的条件判断是否提取为布尔方法

### 6.3 常见反模式

```csharp
// ❌ 反模式1：魔法数字
if (status == 1) { ... }  // 1是什么意思？
// ✅ 改进：使用枚举或常量
if (status == OrderStatus.Paid) { ... }

// ❌ 反模式2：过深的嵌套
if (a)
{
    if (b)
    {
        if (c)
        {
            // 太深了！
        }
    }
}
// ✅ 改进：提前返回
if (!a) return;
if (!b) return;
if (!c) return;
// 正常逻辑

// ❌ 反模式3：复杂的条件
if ((x > 0 && y < 10) || (z == 5 && w != null)) { ... }
// ✅ 改进：提取为有意义的方法
if (IsValidOperation(x, y, z, w)) { ... }
```

**题目**：实现一个循环计算器，支持加减乘除运算。用户输入两个数字和一个运算符，输出结果。输入'q'退出。

```csharp
public void RunCalculator()
{
    Console.WriteLine("=== 简易计算器 ===");
    Console.WriteLine("支持运算: + - * /");
    Console.WriteLine("输入 'q' 退出\n");

    while (true)
    {
        // 获取第一个数字
        Console.Write("输入第一个数字 (或q退出): ");
        string input1 = Console.ReadLine();
        if (input1?.ToLower() == "q") break;

        if (!double.TryParse(input1, out double num1))
        {
            Console.WriteLine("无效的数字!\n");
            continue;
        }

        // 获取运算符
        Console.Write("输入运算符 (+ - * /): ");
        string op = Console.ReadLine();

        // 获取第二个数字
        Console.Write("输入第二个数字: ");
        string input2 = Console.ReadLine();
        if (!double.TryParse(input2, out double num2))
        {
            Console.WriteLine("无效的数字!\n");
            continue;
        }

        // 计算结果
        double result = 0;
        bool valid = true;

        switch (op)
        {
            case "+":
                result = num1 + num2;
                break;
            case "-":
                result = num1 - num2;
                break;
            case "*":
                result = num1 * num2;
                break;
            case "/":
                if (num2 == 0)
                {
                    Console.WriteLine("错误: 除数不能为零!\n");
                    valid = false;
                }
                else
                {
                    result = num1 / num2;
                }
                break;
            default:
                Console.WriteLine($"错误: 不支持的运算符 '{op}'\n");
                valid = false;
                break;
        }

        if (valid)
        {
            Console.WriteLine($"\n结果: {num1} {op} {num2} = {result}\n");
        }
    }

    Console.WriteLine("计算器已关闭");
}
```

### 练习5：综合挑战 - 猜单词游戏
**题目**：实现一个简单的猜单词游戏：
1. 从预设列表中随机选择一个英文单词
2. 显示单词长度的下划线（如 "_ _ _ _ _"）
3. 玩家每次猜一个字母
4. 猜对的字母显示出来，猜错扣生命值
5. 全部猜出或生命值为0时游戏结束

<details>
<summary>查看答案</summary>

```csharp
public class WordGuessingGame
{
    private readonly string[] Words = { "APPLE", "BANANA", "ORANGE", "GRAPE",
                                       "COMPUTER", "PROGRAM", "KEYBOARD", "MONITOR" };

    public void Play()
    {
        Console.WriteLine("=== 猜单词游戏 ===\n");

        Random random = new Random();
        string targetWord = Words[random.Next(Words.Length)];
        char[] guessedLetters = new char[targetWord.Length];
        HashSet<char> guessedSet = new HashSet<char>();
        int lives = 6;
        bool won = false;

        // 初始化显示（全部为下划线）
        for (int i = 0; i < guessedLetters.Length; i++)
        {
            guessedLetters[i] = '_';
        }

        Console.WriteLine($"提示: 这是一个{targetWord.Length}个字母的英文单词");
        Console.WriteLine($"你有{lives}次机会\n");

        while (lives > 0 && !won)
        {
            // 显示当前状态
            Console.WriteLine($"单词: {string.Join(" ", guessedLetters)}");
            Console.WriteLine($"已猜字母: [{string.Join(", ", guessedSet.OrderBy(c => c))}]");
            Console.WriteLine($"剩余生命: {new string('♥', lives)}{new string('♡', 6 - lives)}");
            Console.Write("\n请猜一个字母: ");

            // 模拟输入
            char guess = GetRandomLetter(guessedSet);
            Console.WriteLine(char.ToUpper(guess));
            guess = char.ToUpper(guess);

            // 检查是否已经猜过
            if (guessedSet.Contains(guess))
            {
                Console.WriteLine("这个字母已经猜过了！\n");
                continue;
            }

            guessedSet.Add(guess);

            // 检查字母是否在单词中
            bool found = false;
            for (int i = 0; i < targetWord.Length; i++)
            {
                if (targetWord[i] == guess)
                {
                    guessedLetters[i] = guess;
                    found = true;
                }
            }

            if (found)
            {
                Console.WriteLine($"✓ 正确! 字母 '{guess}' 在单词中\n");

                // 检查是否赢了
                if (!guessedLetters.Contains('_'))
                {
                    won = true;
                }
            }
            else
            {
                lives--;
                Console.WriteLine($"✗ 错误! 字母 '{guess}' 不在单词中");
                Console.WriteLine($"剩余机会: {lives}\n");
            }
        }

        // 游戏结束
        Console.WriteLine("\n====================");
        if (won)
        {
            Console.WriteLine($"🎉 恭喜你猜对了！答案是: {targetWord}");
            Console.WriteLine($"剩余生命: {lives}");
        }
        else
        {
            Console.WriteLine($"😢 游戏结束！正确答案是: {targetWord}");
            Console.WriteLine($"完整单词: {targetWord}");
        }
        Console.WriteLine("====================");
    }

    private char GetRandomLetter(HashSet<char> guessed)
    {
        // 为了演示，生成合理的猜测
        string alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZ";
        Random random = new Random();

        char letter;
        do
        {
            letter = alphabet[random.Next(alphabet.Length)];
        } while (guessed.Contains(letter));

        return letter;
    }
}
```
